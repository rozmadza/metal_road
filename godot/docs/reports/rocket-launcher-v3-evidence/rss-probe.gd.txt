var game:Survival
var images:Array=[]
var records:Array=[]
var samples:Array[float]=[]
var sampling:=false
var last_tick:=0

func _initialize() -> void:run.call_deferred()
func _process(_delta:float) -> bool:
	var now:=Time.get_ticks_usec()
	if sampling and last_tick>0:samples.append(float(now-last_tick)/1000)
	last_tick=now
	return false
func mem() -> Dictionary:
	return {"static":Performance.get_monitor(Performance.MEMORY_STATIC),"video":Performance.get_monitor(Performance.RENDER_VIDEO_MEM_USED),"resources":Performance.get_monitor(Performance.OBJECT_RESOURCE_COUNT),"nodes":Performance.get_monitor(Performance.OBJECT_NODE_COUNT)}
func snap(label:String) -> void:
	await RenderingServer.frame_post_draw
	images.append({"name":label,"image":root.get_texture().get_image()})
	var bs:Array=[]
	for b in game.bullets.get_children():
		if b is Bullet:bs.append({"age":b.age,"angle":b.rotation,"detonated":b.detonated,"owned_by_vfx":b._explosion_owned_by_vfx})
	records.append({"label":label,"bullets":bs,"events":game.get_node("CombatVFX").events.size()})
func freeze_cars() -> void:
	game.set_process(false);game.set_physics_process(false)
	for car in get_nodes_in_group("combatants"):
		car.freeze=true;car.set_physics_process(false);car.gun.set_physics_process(false)
		var ai:=car.get_node_or_null("ArenaBot")
		if ai:ai.set_physics_process(false)
func restart() -> float:
	var start:=Time.get_ticks_usec()
	game._on_retry_requested()
	await process_frame;await physics_frame;await process_frame
	freeze_cars()
	return float(Time.get_ticks_usec()-start)/1000

	for car in get_nodes_in_group("combatants"):car.position=Vector2(1000,300)
	game.player.position=Vector2(450,830);game.player.rotation=0
	cfg=load("res://resources/weapons/rocket_launcher.tres").duplicate();profile=cfg.visual_profile

		cfg.homing_strength=1.25 if args[2]=="baseline" else (0.0 if args[2]=="A" else .3)
