 timeout 180s flatpak run --filesystem=/tmp/destroyed-wreck-after org.godotengine.Godot --headless --path /tmp/destroyed-wreck-after --script "res://tests/$name.gd" > "/tmp/destroyed-wreck-evidence/extra/$name.log" 2>&1
